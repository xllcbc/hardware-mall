### Task 2: 测试 + 构建 workflow

**Files:** Create `.github/workflows/test.yml`

```yaml
name: test-and-build
on:
  pull_request: { branches: [main] }
  push: { branches: [main, 'phase*'] }

jobs:
  backend-test:
    runs-on: ubuntu-latest
    services:
      mysql:
        image: mysql:8.0
        env: { MYSQL_ROOT_PASSWORD: root, MYSQL_DATABASE: hardware_mall_test }
        ports: ['3306:3306']
      redis: { image: redis:7-alpine, ports: ['6379:6379'] }
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with: { distribution: temurin, java-version: 17, cache: maven }
      - name: Run tests
        working-directory: hardware-mall-backend
        env: { SPRING_PROFILES_ACTIVE: test, JWT_SECRET: ${{ secrets.JWT_SECRET_FOR_TEST }}, ADMIN_PASSWORD: TestStrongPwd_2026 }
        run: mvn -B clean test

  admin-build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 18 }
      - working-directory: hardware-mall-admin
        run: npm ci && npm run build

  uniapp-build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 18 }
      - working-directory: hardware-mall-uniapp
        run: npm ci && npm run build:mp-weixin
```

### Task 3: 部署 workflow

**Files:** Create `.github/workflows/backend-admin-deploy.yml`

```yaml
name: deploy-prod
on: { push: { branches: [main] }, workflow_dispatch: }

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: docker/setup-buildx-action@v3
      - uses: docker/login-action@v3
        with: { registry: ${{ secrets.DOCKER_REGISTRY }}, username: ${{ secrets.DOCKER_USERNAME }}, password: ${{ secrets.DOCKER_PASSWORD }} }

      - uses: docker/build-push-action@v5
        with: { context: ./hardware-mall-backend, push: true, tags: ${{ secrets.DOCKER_REGISTRY }}/hardware-mall-backend:${{ github.sha }} }

      - uses: docker/build-push-action@v5
        with: { context: ./hardware-mall-admin, push: true, tags: ${{ secrets.DOCKER_REGISTRY }}/hardware-mall-admin:${{ github.sha }} }

      - uses: appleboy/ssh-action@v1.0.0
        with: { host: ${{ secrets.DEPLOY_HOST }}, username: ${{ secrets.DEPLOY_USER }}, key: ${{ secrets.DEPLOY_SSH_KEY }},
          script: |
            cd /opt/hardware-mall &&
            export IMAGE_TAG=${{ github.sha }} &&
            docker compose -f docker-compose.prod.yml pull &&
            docker compose -f docker-compose.prod.yml up -d
        }
```

### Task 4: uniapp 体验版 CI

**Files:** Create `.github/workflows/uniapp-experience.yml`

```yaml
name: deploy-uniapp-experience
on: { workflow_dispatch: {}, push: { branches: [main], paths: ['hardware-mall-uniapp/**'] } }

jobs:
  upload-experience:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 18 }
      - working-directory: hardware-mall-uniapp
        run: npm ci && npm install miniprogram-ci --no-save && npm run build:mp-weixin
      - name: Upload
        working-directory: hardware-mall-uniapp
        env: { WEAPP_APPID: ${{ secrets.WEAPP_APPID }}, WEAPP_PRIVATE_KEY_PATH: /tmp/private.key, CI_MODE: experience }
        run: |
          echo "${{ secrets.WEAPP_PRIVATE_KEY }}" > /tmp/private.key
          chmod 600 /tmp/private.key
          node scripts/build-uni.js
```

### Task 5: GitHub Secrets 清单（用户手工配置）

| Secret 名 | 用途 |
|---|---|
| `DOCKER_REGISTRY` | 镜像仓库地址 |

| `WEAPP_PRIVATE_KEY` | 上传密钥 |
| `JWT_SECRET_FOR_TEST` | CI 测试用 JWT secret（可选） |
